import os
import wave
from google import genai
from google.genai import types

client = genai.Client(api_key=os.environ.get("GEMINI_API_KEY"))

response = client.models.generate_content(
    model="gemini-2.5-flash-preview-tts",
    contents="Namaste Boss, main NOVA hoon. Aaj mausam bahut accha hai.",
    config=types.GenerateContentConfig(
        response_modalities=["AUDIO"],
        speech_config=types.SpeechConfig(
            voice_config=types.VoiceConfig(
                prebuilt_voice_config=types.PrebuiltVoiceConfig(
                    voice_name="Charon"
                )
            )
        )
    )
)

audio_data = response.candidates[0].content.parts[0].inline_data.data

# Bytes ensure karo
if isinstance(audio_data, str):
    import base64
    audio_data = base64.b64decode(audio_data)

print("Audio bytes:", len(audio_data))

# WAV header ke saath save karo
# Gemini TTS: 24kHz, 16-bit, mono PCM
with wave.open("gemini_voice.wav", "wb") as wf:
    wf.setnchannels(1)
    wf.setsampwidth(2)      # 16-bit
    wf.setframerate(24000)  # 24kHz
    wf.writeframes(audio_data)

print("Saved: gemini_voice.wav")
print("Size:", os.path.getsize("gemini_voice.wav"), "bytes")
os.startfile("gemini_voice.wav")
